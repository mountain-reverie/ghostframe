//! `--text-grid --drm-direct` mode.
//!
//! Same glyphs as `text_grid.rs` but painted into a DRM dumb buffer
//! instead of via X11. Used by e2e when Xorg-on-VKMS doesn't yield
//! PalRle-feasible content. Pixel format XRGB8888 little-endian, same
//! convention as `drm_direct.rs`.

use std::fs::File;
use std::os::fd::{AsFd, BorrowedFd};
use std::thread;
use std::time::Duration;

use drm::buffer::DrmFourcc;
use drm::control::{connector, crtc, Device as ControlDevice};
use drm::Device;

use crate::font::{pixel_set, GLYPH_H, GLYPH_W};
use crate::text_grid::{BG_PIXEL, FG_PIXEL, ORIGIN_X, ORIGIN_Y, TEXT};

struct Card(File);
impl AsFd for Card { fn as_fd(&self) -> BorrowedFd<'_> { self.0.as_fd() } }
impl Device for Card {}
impl ControlDevice for Card {}

const REPAINT_INTERVAL: Duration = Duration::from_millis(33); // ~30 Hz, beats capture rate

pub fn run(card_path: &str) -> Result<(), Box<dyn std::error::Error>> {
    let file = File::options().read(true).write(true).open(card_path)
        .map_err(|e| format!("open {card_path}: {e}"))?;
    let card = Card(file);

    // (Same DRM-master + dumb-buffer setup as drm_direct.rs::run —
    // copy the body up through obtaining `(width, height, buf_handle, mmap)`,
    // then diverge from here.)

    let (width, height, mut mmap) = setup_dumb_scanout(&card)?;
    let stride = width as usize * 4;

    loop {
        // Background fill — pack BG_PIXEL into every pixel.
        for y in 0..height as usize {
            for x in 0..width as usize {
                let off = y * stride + x * 4;
                mmap[off..off + 4].copy_from_slice(&BG_PIXEL.to_le_bytes());
            }
        }
        // Glyphs.
        for (n, ch) in TEXT.chars().enumerate() {
            let gx0 = ORIGIN_X + n as u32 * GLYPH_W;
            let gy0 = ORIGIN_Y;
            for py in 0..GLYPH_H {
                for px in 0..GLYPH_W {
                    if pixel_set(ch, px, py) {
                        let ax = (gx0 + px) as usize;
                        let ay = (gy0 + py) as usize;
                        if ax < width as usize && ay < height as usize {
                            let off = ay * stride + ax * 4;
                            mmap[off..off + 4].copy_from_slice(&FG_PIXEL.to_le_bytes());
                        }
                    }
                }
            }
        }

        thread::sleep(REPAINT_INTERVAL);
    }
}

```wgsl
// Debug-only compute pipeline: writes a known gradient into the
// framebuffer texture so __readPixel can be independently verified.
// NOT used in production rendering — only `__readGradientGolden()`
// dispatches this pipeline.

@group(0) @binding(0) var framebuffer: texture_storage_2d<rgba8unorm, write>;

@compute @workgroup_size(8, 8, 1)
fn debug_gradient(@builtin(global_invocation_id) gid: vec3<u32>) {
    let dims = textureDimensions(framebuffer);
    if (gid.x >= dims.x || gid.y >= dims.y) { return; }
    let r = f32(gid.x & 0xFFu) / 255.0;
    let g = f32(gid.y & 0xFFu) / 255.0;
    textureStore(framebuffer, vec2<i32>(i32(gid.x), i32(gid.y)),
                 vec4<f32>(r, g, 0.0, 1.0));
}
```

```typescript
// Debug-only: dispatch the debug_gradient.wgsl compute pipeline to write a
// known gradient (r=x&255, g=y&255, b=0, a=255) into the framebuffer, then
// read back at 16 sample points and report mismatches. Used by
// e2e_readpixel_correctness to verify __readPixel before any other E2E
// pixel assertion is trusted.
(window as any).__readGradientGolden = async (): Promise<{
  ok: boolean;
  mismatches: Array<{ x: number; y: number; got: number[]; want: number[] }>;
}> => {
  const shaderSrc = await fetch('/webgpu/shaders/debug_gradient.wgsl').then(r => r.text());
  const module = device.createShaderModule({ code: shaderSrc });
  const pipeline = device.createComputePipeline({
    layout: 'auto',
    compute: { module, entryPoint: 'debug_gradient' },
  });
  const bindGroup = device.createBindGroup({
    layout: pipeline.getBindGroupLayout(0),
    entries: [{ binding: 0, resource: framebufferTexture.createView() }],
  });
  const encoder = device.createCommandEncoder();
  const pass = encoder.beginComputePass();
  pass.setPipeline(pipeline);
  pass.setBindGroup(0, bindGroup);
  const fbW = framebufferTexture.width;
  const fbH = framebufferTexture.height;
  pass.dispatchWorkgroups(Math.ceil(fbW / 8), Math.ceil(fbH / 8), 1);
  pass.end();
  device.queue.submit([encoder.finish()]);
  await device.queue.onSubmittedWorkDone();

  const samplePoints: Array<[number, number]> = [
    [0, 0], [1, 0], [255, 0], [256, 0],
    [0, 1], [0, 255], [0, 256],
    [100, 200], [320, 240], [640 - 1, 480 - 1],
    [16, 16], [31, 31], [32, 32], [63, 63],
    [Math.floor(fbW / 2), Math.floor(fbH / 2)],
    [fbW - 1, fbH - 1],
  ];
  const mismatches: Array<{ x: number; y: number; got: number[]; want: number[] }> = [];
  for (const [x, y] of samplePoints) {
    if (x >= fbW || y >= fbH) continue;
    const got = await (window as any).__readPixel(x, y);
    const want = [x & 0xFF, y & 0xFF, 0, 255];
    if (got[0] !== want[0] || got[1] !== want[1] || got[2] !== want[2] || got[3] !== want[3]) {
      mismatches.push({ x, y, got, want });
    }
  }
  return { ok: mismatches.length === 0, mismatches };
};
```

//! `--palrle-exact --drm-direct` mode.
//!
//! Paints a deterministic 2-color tile pattern into a DRM dumb buffer so the
//! classifier emits Codec::PalRle with a known palette and known indices.
//! Drives e2e_palrle_exact_pixels for B1 verification.
//!
//! Pattern: 32x32 tiles. Even-column tiles all-color-A; odd-column tiles
//! checkerboard A/B at 1px granularity (forces alternating-nibble RLE).

use std::fs::File;
use std::os::fd::{AsFd, BorrowedFd};
use std::thread;
use std::time::Duration;

use crate::drm_direct::{setup_dumb_scanout, Card};

/// XRGB8888 little-endian: bytes in memory `[B, G, R, X]`.
/// Color A: pure red. Color B: pure blue.
pub const COLOR_A: u32 = 0x0000_0000_FF_00_00; // 0xFF0000 = red in XRGB
pub const COLOR_B: u32 = 0x0000_0000_00_00_FF; // 0x0000FF = blue
// Note: COLOR_A constants above include a leading 0; canonical form is
// `0x00FF0000` (red) and `0x000000FF` (blue). Adjust during compile if
// shorthand confuses; the byte order is what matters.

/// Test verifies these specific (x, y) → expected_rgba mappings.
pub struct ExactSample {
    pub x: u32,
    pub y: u32,
    pub expected_rgba: [u8; 4],
}

pub const SAMPLES: &[ExactSample] = &[
    // Tile (1, 1) — odd column, so checkerboard pattern.

//! `--solid-per-tile --drm-direct` mode.
//!
//! Four 32x32 corner tiles painted a fixed color (top-left red, top-right
//! green, bottom-left blue, bottom-right yellow). A central 64x64 region
//! cycles colors every frame to keep the classifier in tile-codec mode.
//! Everything else is BG dark grey.

use std::fs::File;
use std::os::fd::{AsFd, BorrowedFd};
use std::thread;
use std::time::{Duration, Instant};

use crate::drm_direct::{setup_dumb_scanout, Card};

pub struct CornerSample {
    pub x: u32,
    pub y: u32,
    pub expected_rgba: [u8; 4],
}

/// (x, y) coords are at the CENTER of each corner tile. Coordinates assume
/// the standard 640x480 test capture; if width/height differ, the
/// bottom/right anchors shift but the (16, 16) top-left is invariant.
pub fn samples(width: u32, height: u32) -> Vec<CornerSample> {
    let bx = width.saturating_sub(16);
    let by = height.saturating_sub(16);
    vec![
        CornerSample { x: 16, y: 16, expected_rgba: [0xFF, 0x00, 0x00, 0xFF] },
        CornerSample { x: bx, y: 16, expected_rgba: [0x00, 0xFF, 0x00, 0xFF] },
        CornerSample { x: 16, y: by, expected_rgba: [0x00, 0x00, 0xFF, 0xFF] },
        CornerSample { x: bx, y: by, expected_rgba: [0xFF, 0xFF, 0x00, 0xFF] },
    ]
}

pub fn run(card_path: &str) -> Result<(), Box<dyn std::error::Error>> {
    let file = File::options().read(true).write(true).open(card_path)
        .map_err(|e| format!("open {card_path}: {e}"))?;
    let card = Card(file);
    let (width, height, mut mmap) = setup_dumb_scanout(&card)?;
    let stride = width as usize * 4;
    let bg: u32 = 0x0014_1414;
    let red: u32 = 0x00FF_0000;
    let green: u32 = 0x0000_FF00;
    let blue: u32 = 0x0000_00FF;
    let yellow: u32 = 0x00FF_FF00;
    let start = Instant::now();

    loop {
        // Background.
        for y in 0..height as usize {
            for x in 0..width as usize {
                let off = y * stride + x * 4;
                mmap[off..off + 4].copy_from_slice(&bg.to_le_bytes());
            }
        }
        // Corners (each 32x32).
        for (cx, cy, color) in &[
            (0, 0, red),
            (width as i32 - 32, 0, green),
            (0, height as i32 - 32, blue),
            (width as i32 - 32, height as i32 - 32, yellow),
        ] {
            for dy in 0..32 {
                for dx in 0..32 {
                    let ax = (*cx + dx) as usize;
                    let ay = (*cy + dy) as usize;
                    if ax < width as usize && ay < height as usize {
                        let off = ay * stride + ax * 4;
                        mmap[off..off + 4].copy_from_slice(&color.to_le_bytes());
                    }
                }
            }
        }
        // Motion region: cycle (64x64) hue at the center.
        let phase = (start.elapsed().as_millis() / 33) as u32 & 0xFF;
        let motion_color: u32 = (phase << 16) | (phase.wrapping_add(85) << 8) | phase.wrapping_add(170);
        let mcx = (width / 2 - 32) as usize;
        let mcy = (height / 2 - 32) as usize;
        for dy in 0..64usize {
            for dx in 0..64usize {
                let off = (mcy + dy) * stride + (mcx + dx) * 4;
                mmap[off..off + 4].copy_from_slice(&motion_color.to_le_bytes());
            }
        }
        thread::sleep(Duration::from_millis(33));
    }
}
