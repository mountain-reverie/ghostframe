# M3.3a CDF 5/3 Server-Side Encoder Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Land all server-side and wire-protocol changes for CDF 5/3 — the lossless progressive-refinement codec that replaces Raw for high-color tiles in the GPU classifier path. Client decode lands in M3.3b.

**Architecture:** GPU pipeline gains Stage 4 (indirect-args dispatch: cdf53_compact → cdf53_indirect_args → cdf53_forward at 3 decomposition levels). Phase B (CPU) extracts 14 bit-planes per tile (sign + 13 magnitude planes, channel-grouped, RLE-compressed). Scheduler partitions tick() output between priority (new dirty tiles) and refinement (multi-pass Cdf53) via `refinement_bandwidth_fraction` with adaptive halving on sustained ACK loss. HELLO `supports_cdf53` bit + `GHOSTFRAME_ENABLE_CDF53` env var form a two-layer gate.

**Tech Stack:** Rust, Vulkan (ash bindings), GLSL compute shaders, TypeScript (one-line web-client HELLO emission update), tokio.

```rust
// ghostframe-lib/src/encoder/cdf53.rs
//! CPU reference implementation of CDF 5/3 wavelet codec.
//!
//! This is the algorithmic source of truth for the codec:
//!   - Unit-tests verify forward+inverse roundtrip (lossless).
//!   - M3.3a's Vulkan compute Stage 4 shaders verify GPU=CPU equivalence
//!     against this reference.
//!   - M3.3b's WGSL inverse compute shader is ported from this reference
//!     (NOT compiled to WASM — WebGPU compute is the deployed path).

/// Number of progressive passes emitted per Cdf53 tile.
/// = 1 sign-bit-plane + 13 magnitude bit-planes covering worst-case
/// 14-bit signed coefficients after 3 levels of CDF 5/3 lifting on
/// 8-bit input.
pub const CDF53_PASS_COUNT: usize = 14;

/// Per-tile coefficient count per channel: 1024 (32×32 tile).
pub const CDF53_COEFFS_PER_CHANNEL: usize = 1024;

/// Number of color channels encoded: BGR (alpha dropped, always 0xFF).
pub const CDF53_CHANNELS: usize = 3;

/// Total coefficients per tile = 3072.
pub const CDF53_TOTAL_COEFFS: usize = CDF53_CHANNELS * CDF53_COEFFS_PER_CHANNEL;

/// Per-channel bit-plane byte size = 1024 bits / 8 = 128 bytes.
const BIT_PLANE_BYTES_PER_CHANNEL: usize = CDF53_COEFFS_PER_CHANNEL / 8;
```

```rust
/// Forward CDF 5/3 lifting transform on a 32×32 BGRA tile.
/// Strips alpha and returns 3072 signed integer coefficients
/// in [B-coeffs (1024)][G-coeffs (1024)][R-coeffs (1024)] order.
/// Each channel's coefficients are laid out as:
///   [LL3 (16)][LH3 (16)][HL3 (16)][HH3 (16)]
///   [LH2 (64)][HL2 (64)][HH2 (64)]
///   [LH1 (256)][HL1 (256)][HH1 (256)]
pub fn forward(tile_bgra: &[u8]) -> Vec<i16> {
    assert_eq!(tile_bgra.len(), 32 * 32 * 4, "tile must be 32×32 BGRA");
    let mut output = vec![0i16; CDF53_TOTAL_COEFFS];

    for ch in 0..CDF53_CHANNELS {
        // Load channel into a 32×32 working buffer (i32 to absorb growth).
        let mut work = [[0i32; 32]; 32];
        for y in 0..32 {
            for x in 0..32 {
                let idx = (y * 32 + x) * 4 + ch;
                work[y][x] = tile_bgra[idx] as i32;
            }
        }

        // 3 levels of 2D decomposition.
        let mut size = 32;
        for _level in 0..3 {
            // Horizontal pass on the size×size LL subband.
            for y in 0..size {
                row_forward(&mut work[y], size);
            }
            // Vertical pass.
            for x in 0..size {
                let mut column = [0i32; 32];
                for y in 0..size {
                    column[y] = work[y][x];
                }
                col_forward(&mut column, size);
                for y in 0..size {
                    work[y][x] = column[y];
                }
            }
            size /= 2;
        }

        // Pack work[][] into channel layout. After 3 levels:
        //   work[0..4][0..4]       = LL3
        //   work[0..4][4..8]       = HL3
        //   work[4..8][0..4]       = LH3
        //   work[4..8][4..8]       = HH3
        //   work[0..8][8..16]      = HL2
        //   work[8..16][0..8]      = LH2
        //   work[8..16][8..16]     = HH2
        //   work[0..16][16..32]    = HL1
        //   work[16..32][0..16]    = LH1
        //   work[16..32][16..32]   = HH1
        let channel_offset = ch * CDF53_COEFFS_PER_CHANNEL;
        let mut out_idx = channel_offset;

        // LL3, LH3, HL3, HH3 (4×4 each → 16 each)
        for y in 0..4 { for x in 0..4 { output[out_idx] = work[y][x] as i16; out_idx += 1; } }       // LL3
        for y in 0..4 { for x in 4..8 { output[out_idx] = work[y][x] as i16; out_idx += 1; } }       // HL3
        for y in 4..8 { for x in 0..4 { output[out_idx] = work[y][x] as i16; out_idx += 1; } }       // LH3
        for y in 4..8 { for x in 4..8 { output[out_idx] = work[y][x] as i16; out_idx += 1; } }       // HH3
        // LH2, HL2, HH2 (8×8 each → 64 each)
        for y in 0..8  { for x in 8..16  { output[out_idx] = work[y][x] as i16; out_idx += 1; } }    // HL2
        for y in 8..16 { for x in 0..8   { output[out_idx] = work[y][x] as i16; out_idx += 1; } }    // LH2
        for y in 8..16 { for x in 8..16  { output[out_idx] = work[y][x] as i16; out_idx += 1; } }    // HH2
        // LH1, HL1, HH1 (16×16 each → 256 each)
        for y in 0..16  { for x in 16..32 { output[out_idx] = work[y][x] as i16; out_idx += 1; } }   // HL1
        for y in 16..32 { for x in 0..16  { output[out_idx] = work[y][x] as i16; out_idx += 1; } }   // LH1
        for y in 16..32 { for x in 16..32 { output[out_idx] = work[y][x] as i16; out_idx += 1; } }   // HH1
        assert_eq!(out_idx, channel_offset + CDF53_COEFFS_PER_CHANNEL);
    }

    output
}

/// CDF 5/3 lifting on a single row (or column) of length `n` (must be even).
/// Operates in-place. Lifting steps:
///   1. predict: x[2i+1] -= (x[2i] + x[2i+2] + 1) / 2     (handle boundary with mirror)
///   2. update : x[2i]   += (x[2i-1] + x[2i+1] + 2) / 4   (mirror at boundaries)
/// After lifting, even indices hold low-pass (L) and odd indices hold high-pass (H).
/// We then re-arrange to [L_0..L_{n/2-1}, H_0..H_{n/2-1}] (deinterleave).
fn row_forward(buf: &mut [i32], n: usize) {
    // Predict step (odd indices become high-pass).
    for i in (1..n).step_by(2) {
        let left = buf[i - 1];
        let right = if i + 1 < n { buf[i + 1] } else { buf[i - 1] }; // mirror
        buf[i] -= (left + right) >> 1;
    }
    // Update step (even indices become low-pass).
    for i in (0..n).step_by(2) {
        let left = if i == 0 { buf[1] } else { buf[i - 1] };       // mirror
        let right = if i + 1 < n { buf[i + 1] } else { buf[i - 1] };// mirror
        buf[i] += (left + right + 2) >> 2;
    }
    // Deinterleave: L L L L H H H H
    let mut tmp = vec![0i32; n];
    let half = n / 2;
    for i in 0..half {
        tmp[i] = buf[2 * i];
        tmp[half + i] = buf[2 * i + 1];
    }
    buf[..n].copy_from_slice(&tmp);
}

fn col_forward(buf: &mut [i32], n: usize) {
    row_forward(buf, n);
}
```

```rust
/// Inverse CDF 5/3 lifting transform. Takes 3072 coefficients
/// (3 channels × 1024) and returns the recovered 32×32 BGR tile
/// as 3072 bytes (1024 pixels × 3 channels, no alpha).
pub fn inverse(coefficients: &[i16]) -> Vec<u8> {
    assert_eq!(coefficients.len(), CDF53_TOTAL_COEFFS);
    let mut output = vec![0u8; 32 * 32 * 3];

    for ch in 0..CDF53_CHANNELS {
        // Unpack channel layout back to work[][].
        let mut work = [[0i32; 32]; 32];
        let channel_offset = ch * CDF53_COEFFS_PER_CHANNEL;
        let mut in_idx = channel_offset;

        for y in 0..4 { for x in 0..4 { work[y][x] = coefficients[in_idx] as i32; in_idx += 1; } }       // LL3
        for y in 0..4 { for x in 4..8 { work[y][x] = coefficients[in_idx] as i32; in_idx += 1; } }       // HL3
        for y in 4..8 { for x in 0..4 { work[y][x] = coefficients[in_idx] as i32; in_idx += 1; } }       // LH3
        for y in 4..8 { for x in 4..8 { work[y][x] = coefficients[in_idx] as i32; in_idx += 1; } }       // HH3
        for y in 0..8  { for x in 8..16  { work[y][x] = coefficients[in_idx] as i32; in_idx += 1; } }
        for y in 8..16 { for x in 0..8   { work[y][x] = coefficients[in_idx] as i32; in_idx += 1; } }
        for y in 8..16 { for x in 8..16  { work[y][x] = coefficients[in_idx] as i32; in_idx += 1; } }
        for y in 0..16  { for x in 16..32 { work[y][x] = coefficients[in_idx] as i32; in_idx += 1; } }
        for y in 16..32 { for x in 0..16  { work[y][x] = coefficients[in_idx] as i32; in_idx += 1; } }
        for y in 16..32 { for x in 16..32 { work[y][x] = coefficients[in_idx] as i32; in_idx += 1; } }
        assert_eq!(in_idx, channel_offset + CDF53_COEFFS_PER_CHANNEL);

        // 3 levels of 2D inverse decomposition.
        let mut size = 8;
        for _level in 0..3 {
            // Vertical inverse.
            for x in 0..size {
                let mut column = [0i32; 32];
                for y in 0..size {
                    column[y] = work[y][x];
                }
                col_inverse(&mut column, size);
                for y in 0..size {
                    work[y][x] = column[y];
                }
            }
            // Horizontal inverse.
            for y in 0..size {
                row_inverse(&mut work[y], size);
            }
            size *= 2;
        }

        // Pack to output[][] in BGR order with stride 3.
        for y in 0..32 {
            for x in 0..32 {
                let v = work[y][x].clamp(0, 255) as u8;
                output[(y * 32 + x) * 3 + ch] = v;
            }
        }
    }

    output
}

fn row_inverse(buf: &mut [i32], n: usize) {
    let half = n / 2;
    // Re-interleave: L0 H0 L1 H1 ...
    let mut tmp = vec![0i32; n];
    for i in 0..half {
        tmp[2 * i] = buf[i];
        tmp[2 * i + 1] = buf[half + i];
    }
    buf[..n].copy_from_slice(&tmp);
    // Inverse update.
    for i in (0..n).step_by(2) {
        let left = if i == 0 { buf[1] } else { buf[i - 1] };
        let right = if i + 1 < n { buf[i + 1] } else { buf[i - 1] };
        buf[i] -= (left + right + 2) >> 2;
    }
    // Inverse predict.
    for i in (1..n).step_by(2) {
        let left = buf[i - 1];
        let right = if i + 1 < n { buf[i + 1] } else { buf[i - 1] };
        buf[i] += (left + right) >> 1;
    }
}

fn col_inverse(buf: &mut [i32], n: usize) {
    row_inverse(buf, n);
}
```

```rust
/// Encode a tile's coefficients into 14 progressive bit-plane passes.
/// Each pass payload is the concatenation of 3 channels' RLE-encoded
/// bit-planes in BGR order:
///   [u16 BE len_B] [len_B bytes RLE bit-plane B]
///   [u16 BE len_G] [len_G bytes RLE bit-plane G]
///   [u16 BE len_R] [len_R bytes RLE bit-plane R]
/// Pass 0 = sign-bit-plane (bit set iff coefficient is negative).
/// Pass 1..13 = magnitude bit-planes from bit 12 down to bit 0.
pub fn encode_passes(coefficients: &[i16]) -> Vec<Vec<u8>> {
    assert_eq!(coefficients.len(), CDF53_TOTAL_COEFFS);
    let mut passes = Vec::with_capacity(CDF53_PASS_COUNT);
    for pass_idx in 0..CDF53_PASS_COUNT {
        let mut payload = Vec::new();
        for ch in 0..CDF53_CHANNELS {
            let channel_offset = ch * CDF53_COEFFS_PER_CHANNEL;
            let channel = &coefficients[channel_offset..channel_offset + CDF53_COEFFS_PER_CHANNEL];
            let bit_plane = extract_bit_plane(channel, pass_idx);
            let rle = rle_encode(&bit_plane);
            let len = rle.len() as u16;
            payload.push((len >> 8) as u8);
            payload.push(len as u8);
            payload.extend_from_slice(&rle);
        }
        passes.push(payload);
    }
    passes
}

/// Extract the bit-plane for the given pass index from a channel's coefficients.
/// Returns 128 bytes (1024 bits) — bit i corresponds to coefficient i.
/// pass_idx 0 = sign plane; pass_idx 1..=13 = magnitude bit 12..0.
fn extract_bit_plane(channel: &[i16], pass_idx: usize) -> Vec<u8> {
    debug_assert_eq!(channel.len(), CDF53_COEFFS_PER_CHANNEL);
    let mut out = vec![0u8; BIT_PLANE_BYTES_PER_CHANNEL];
    for (i, &coeff) in channel.iter().enumerate() {
        let bit = if pass_idx == 0 {
            // Sign plane.
            (coeff < 0) as u8
        } else {
            // Magnitude bit (13 - (pass_idx - 1)) = 14 - pass_idx, 0-indexed from LSB.
            // pass_idx=1 → bit 13, ..., pass_idx=13 → bit 1, pass_idx=14 → bit 0.
            // Wait: 14 passes total, 1 sign + 13 magnitude.
            // pass 1 → MSB bit 12, pass 13 → LSB bit 0.
            let bit_pos = 13 - pass_idx; // pass 1 → 12, pass 13 → 0
            let magnitude = coeff.unsigned_abs() as u32;
            ((magnitude >> bit_pos) & 1) as u8
        };
        if bit != 0 {
            out[i / 8] |= 1 << (i % 8);
        }
    }
    out
}

/// Run-length encode a byte buffer. Token format:
///   - 0x00..=0x7F: literal data byte.
///   - 0x80..=0xFF: zero run of (token & 0x7F) + 1 bytes (1..=128 zeros).
/// Optimized for sparse bit-planes where most bytes are zero.
fn rle_encode(data: &[u8]) -> Vec<u8> {
    let mut out = Vec::new();
    let mut i = 0;
    while i < data.len() {
        if data[i] == 0 {
            // Count run of zeros, up to 128.
            let mut run_len = 1;
            while i + run_len < data.len() && data[i + run_len] == 0 && run_len < 128 {
                run_len += 1;
            }
            out.push(0x80 | ((run_len - 1) as u8));
            i += run_len;
        } else if data[i] < 0x80 {
            out.push(data[i]);
            i += 1;
        } else {
            // High-bit-set literal — escape: emit run-of-zero=1 byte to indicate
            // "next byte is literal even if high bit is set". Simpler: pre-shift
            // by storing nibbles. For this codec, bit-plane bytes are 0..0xFF
            // (8 bits packed), so we need to handle 0x80..0xFF literals.
            // Switch to a two-byte literal escape: 0x7F as marker + raw byte.
            out.push(0x7F);
            out.push(data[i]);
            i += 1;
        }
    }
    out
}

fn rle_decode(rle: &[u8]) -> Vec<u8> {
    let mut out = Vec::new();
    let mut i = 0;
    while i < rle.len() {
        let token = rle[i];
        if token == 0x7F {
            // Two-byte literal escape.
            if i + 1 < rle.len() {
                out.push(rle[i + 1]);
                i += 2;
            } else {
                break;
            }
        } else if token & 0x80 != 0 {
            // Zero run.
            let run_len = (token & 0x7F) as usize + 1;
            out.extend(std::iter::repeat(0u8).take(run_len));
            i += 1;
        } else {
            out.push(token);
            i += 1;
        }
    }
    out
}
```

```rust
/// Reassemble coefficients from N bit-plane passes (N ≤ CDF53_PASS_COUNT).
/// Truncated K < CDF53_PASS_COUNT yields a lossy approximation; passing all
/// CDF53_PASS_COUNT passes recovers the exact original coefficients.
pub fn decode_passes(passes: &[&[u8]]) -> Vec<i16> {
    assert!(passes.len() <= CDF53_PASS_COUNT);
    let mut coefficients = vec![0i16; CDF53_TOTAL_COEFFS];
    // Magnitude accumulation: u32 magnitude per coefficient.
    let mut magnitudes = vec![0u32; CDF53_TOTAL_COEFFS];
    // Sign tracking: separate from magnitude until final assembly.
    let mut signs = vec![false; CDF53_TOTAL_COEFFS];

    for (pass_idx, payload) in passes.iter().enumerate() {
        // Parse the per-pass payload (3 channels of RLE-encoded bit-planes).
        let mut offset = 0;
        for ch in 0..CDF53_CHANNELS {
            if offset + 2 > payload.len() { return coefficients; } // malformed; bail
            let len = ((payload[offset] as u16) << 8) | (payload[offset + 1] as u16);
            offset += 2;
            if offset + len as usize > payload.len() { return coefficients; }
            let rle = &payload[offset..offset + len as usize];
            offset += len as usize;
            let bit_plane = rle_decode(rle);
            let channel_offset = ch * CDF53_COEFFS_PER_CHANNEL;

            for i in 0..CDF53_COEFFS_PER_CHANNEL {
                let byte = bit_plane.get(i / 8).copied().unwrap_or(0);
                let bit = (byte >> (i % 8)) & 1;
                if bit != 0 {
                    if pass_idx == 0 {
                        signs[channel_offset + i] = true;
                    } else {
                        let bit_pos = 13 - pass_idx;
                        magnitudes[channel_offset + i] |= 1u32 << bit_pos;
                    }
                }
            }
        }
    }

    for i in 0..CDF53_TOTAL_COEFFS {
        let mag = magnitudes[i].min(i16::MAX as u32) as i16;
        coefficients[i] = if signs[i] { -mag } else { mag };
    }

    coefficients
}
```

```rust
            // combinations within a tile, far exceeding 16.

            1,
            0,
            y as i16,
```

```rust
//! HELLO message (msg_type = 0x03): client → server one-shot
//! capability advertisement on the reliable bidi FEEDBACK stream.
//!
//! Wire format (2 bytes):
//!   [0]  msg_type = 0x03
//!   [1]  capabilities : u8   bit 0 = supports indices_raw (PalRle flags bit 1)
//!                            bit 1 = supports cdf53 (high-color tile codec)
//!                            bits 2..7 reserved (must be 0; deserializer ignores)

pub const HELLO_MSG_TYPE: u8 = 0x03;
pub const HELLO_SIZE: usize = 2;

#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
pub struct ClientCapabilities {
    pub indices_raw_enabled: bool,
    pub supports_cdf53: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct HelloMsg {
    pub caps: ClientCapabilities,
}

impl HelloMsg {
    pub fn decode(data: &[u8]) -> Option<Self> {
        if data.len() < HELLO_SIZE { return None; }
        if data[0] != HELLO_MSG_TYPE { return None; }
        let caps_byte = data[1];
        Some(Self {
            caps: ClientCapabilities {
                indices_raw_enabled: (caps_byte & 0x01) != 0,
                supports_cdf53:      (caps_byte & 0x02) != 0,
            },
        })
    }

    pub fn encode(&self, buf: &mut Vec<u8>) {
        buf.push(HELLO_MSG_TYPE);
        let mut caps_byte = 0u8;
        if self.caps.indices_raw_enabled { caps_byte |= 0x01; }
        if self.caps.supports_cdf53      { caps_byte |= 0x02; }
        buf.push(caps_byte);
    }
}
```

Replace the existing `pub fn tick(...)` with a version that partitions:

```rust
    pub fn tick(&mut self, budget_bytes: usize) -> Vec<TileWork> {
        let fraction = self.refinement_bandwidth_fraction;
        let mut refinement_budget = (budget_bytes as f32 * fraction) as usize;
        let mut priority_budget = budget_bytes.saturating_sub(refinement_budget);

        // Empty-queue repurposing: if one queue is empty, give all budget
        // to the other (no wasted bandwidth).
        if self.refinement_queue.is_empty() {
            priority_budget = budget_bytes;
            refinement_budget = 0;
        } else if self.priority_queue.is_empty() {
            refinement_budget = budget_bytes;
            priority_budget = 0;
        }

        let mut emitted = Vec::new();

        // Drain priority queue first (new dirty tiles take precedence within
        // their budget share).
        Self::drain_queue(&mut self.priority_queue, priority_budget, &mut emitted);

        // Then drain refinement queue (pass-major order is maintained by
        // round-robin among entries at the same pass_idx). We need to emit
        // by ascending pass_idx so that pass 0 of every tile goes before
        // pass 1 of any tile.
        Self::drain_refinement_pass_major(&mut self.refinement_queue, refinement_budget, &mut emitted);

        self.delivery_window_emitted = self.delivery_window_emitted.saturating_add(emitted.len() as u32);
        self.maybe_adjust_refinement_fraction();

        emitted
    }

    fn drain_queue(queue: &mut VecDeque<TileWork>, budget: usize, out: &mut Vec<TileWork>) {
        let mut spent = 0usize;
        while let Some(front) = queue.front() {
            let cost = front.payload.len();
            if spent + cost > budget { break; }
            let mut work = queue.pop_front().unwrap();
            work.last_sent_at = Some(Instant::now());
            work.state = WorkState::InFlight;
            spent += cost;
            out.push(work);
        }
    }

    /// Drain refinement queue in pass-major order: every tile's pass 0 before
    /// any tile's pass 1. Implementation: find min pass_idx in the queue,
    /// drain all entries at that pass_idx (round-robin among tiles), then
    /// repeat for next pass_idx — all under the budget cap.
    fn drain_refinement_pass_major(
        queue: &mut VecDeque<TileWork>,
        budget: usize,
        out: &mut Vec<TileWork>,
    ) {
        let mut spent = 0usize;
        loop {
            // Find the smallest pass_idx still in queue.
            let min_pass = match queue.iter().map(|w| w.pass_idx).min() {
                Some(p) => p,
                None => break,
            };
            // Drain all entries at this pass_idx in insertion order, subject to budget.
            let mut still_at_this_pass = false;
            let mut idx = 0;
            while idx < queue.len() {
                if queue[idx].pass_idx == min_pass {
                    let cost = queue[idx].payload.len();
                    if spent + cost > budget {
                        return; // budget exhausted
                    }
                    let mut work = queue.remove(idx).unwrap();
                    work.last_sent_at = Some(Instant::now());
                    work.state = WorkState::InFlight;
                    spent += cost;
                    out.push(work);
                    still_at_this_pass = true;
                } else {
                    idx += 1;
                }
            }
            if !still_at_this_pass { break; }
        }
    }
```

```rust
    /// Called at the end of each tick(). If the rolling delivery rate
    /// (acks / emissions) is sustained low (<0.5 for 10 consecutive
    /// rounds), halve the refinement fraction (floor 0.05). If sustained
    /// high (≥0.8 for 10 rounds), double (ceiling 0.2). Reset window
    /// counters each round.
    fn maybe_adjust_refinement_fraction(&mut self) {
        const ADJUST_AFTER_ROUNDS: u32 = 10;
        const LOW_DELIVERY: f32 = 0.5;
        const HIGH_DELIVERY: f32 = 0.8;
        const MIN_FRACTION: f32 = 0.05;
        const MAX_FRACTION: f32 = 0.2;

        let rate = if self.delivery_window_emitted == 0 {
            1.0  // No emission = no signal; treat as healthy.
        } else {
            self.delivery_window_acked as f32 / self.delivery_window_emitted as f32
        };

        if rate < LOW_DELIVERY {
            self.low_delivery_rounds += 1;
            self.high_delivery_rounds = 0;
            if self.low_delivery_rounds >= ADJUST_AFTER_ROUNDS {
                self.refinement_bandwidth_fraction =
                    (self.refinement_bandwidth_fraction * 0.5).max(MIN_FRACTION);
                self.low_delivery_rounds = 0;
            }
        } else if rate >= HIGH_DELIVERY {
            self.high_delivery_rounds += 1;
            self.low_delivery_rounds = 0;
            if self.high_delivery_rounds >= ADJUST_AFTER_ROUNDS {
                self.refinement_bandwidth_fraction =
                    (self.refinement_bandwidth_fraction * 2.0).min(MAX_FRACTION);
                self.high_delivery_rounds = 0;
            }
        } else {
            // Mid-band: decay both streaks toward 0.
            self.low_delivery_rounds = self.low_delivery_rounds.saturating_sub(1);
            self.high_delivery_rounds = self.high_delivery_rounds.saturating_sub(1);
        }

        // Reset window for next round.
        self.delivery_window_emitted = 0;
        self.delivery_window_acked = 0;
    }

    /// Diagnostic accessor: current refinement-bandwidth fraction. Used by
    /// unit tests and for future tracing/observability.
    pub fn current_refinement_fraction(&self) -> f32 {
        self.refinement_bandwidth_fraction
    }
```

```glsl
// Stage 4a: produce the dense list of (dirty AND high-color) tile indices
// for Cdf53 forward processing.
//
// Predicate: tile is dirty (sad > threshold) AND tile_analysis.count > 16
//            AND tile_analysis.count != UNIQUE_COLORS_UNKNOWN_SENTINEL.
//
// Mirrors palrle_compact.comp but with the inverse color-count condition.
// Dispatch: one workgroup per tile (cols × rows × 1). One thread per workgroup.

#version 450

layout(local_size_x = 1, local_size_y = 1, local_size_z = 1) in;

layout(binding = 0, std430) readonly buffer SadBuffer {
    uint sad[];
};

struct TileAnalysis {
    uint  count;
    uint  edge_density_thou;
    uint  _pad0;
    uint  _pad1;
    uint  colors[16];
};
layout(binding = 1, std430) readonly buffer AnalysisInput {
    TileAnalysis tiles[];
};

layout(binding = 2, std430) buffer Cdf53CompactList {
    uint list[];
};

layout(binding = 3, std430) buffer Cdf53CompactCount {
    uint count;
};

layout(push_constant) uniform PushConstants {
    uint cols;
    uint rows;
    uint dirty_threshold;
    uint unique_colors_unknown_sentinel;
};

void main() {
    uint tile_idx = gl_WorkGroupID.x;
    if (tile_idx >= cols * rows) return;

    uint sad_v = sad[tile_idx];

    if (sad_v > dirty_threshold && c > 16u && c != unique_colors_unknown_sentinel) {

        list[slot] = tile_idx;
    }
```

## Task 7: GPU pipeline `cdf53_forward_l{1,2,3}.comp` — three-level wavelet forward

**Files:**

Three separate shaders for portability — each runs one decomposition level. All dispatch indirectly using `cdf53_dispatch_args` (one workgroup per Cdf53 tile). Workgroup sizes:
- L1: (16, 16, 1) = 256 invocations, covering 32×32 in 4 sub-quadrants — but for portability ceiling we use 1 workgroup per quadrant (4 workgroups per tile via the dispatch_y axis). Actually simpler: one dispatch group handles a 32×32 tile across two 16×16 sub-dispatches via `gl_LocalInvocationID` and `gl_WorkGroupID.y` for the sub-quadrant.

For clarity and to avoid inter-workgroup sync, **each level is one workgroup per tile** with a workgroup_size matching the level's size:

- L1: workgroup_size = (16, 16, 1) = 256 invocations. Each invocation handles a 2×2 block of the 32×32 input. Workgroup writes the 32×32 output (LL1 in top-left quadrant, details in others).
- L2: workgroup_size = (16, 16, 1) = 256. Operates on the 16×16 LL1 region from L1's output. Writes LL2 + details.
- L3: workgroup_size = (8, 8, 1) = 64. Operates on the 8×8 LL2 region from L2's output.

This avoids the 1024-invocation portability cliff.

- [ ] **Step 1: Write the L1 shader**

```glsl
// ghostframe-lib/src/capture/shaders/cdf53_forward_l1.comp
// Stage 4c-L1: CDF 5/3 forward wavelet transform, level 1.
// Each workgroup processes one 32×32 tile (indexed by gl_WorkGroupID.x
// into cdf53_compact_list[]). Workgroup_size = 16×16 = 256 invocations,
// each handling a 2×2 block (predict + update lifting pair).
//
// Input: current BGRA frame (imported texture).
// Output: cdf53_coefficients_buffer[tile_idx][channel][1024 coeffs] —
//         after L1, LL1 occupies the top-left 16×16, details fill the rest.

#version 450

layout(local_size_x = 16, local_size_y = 16, local_size_z = 1) in;

// Workgroup-shared 32×32 scratch buffer per channel. We process 3 channels
// (BGR) serially within the shader to keep shared memory bounded.
shared int scratch[32 * 32];

layout(binding = 0) uniform sampler2D current_frame_sampler;

layout(binding = 1, std430) readonly buffer Cdf53CompactList {
    uint tile_indices[];
};

layout(binding = 2, std430) buffer Cdf53Coefficients {
    int coefficients[];   // i16-equivalent; declared as int (32-bit) here.
                          // Phase B reinterprets via memcpy: lower 16 bits per coeff.
};

layout(push_constant) uniform PushConstants {
    uint cols;         // tile_count_x
    uint rows;         // tile_count_y
    uint tile_size;    // 32
};

const uint COEFFS_PER_CHANNEL = 1024u;
const uint CHANNELS = 3u;

void load_channel_into_scratch(uint tile_x, uint tile_y, uint channel) {
    // Each invocation loads 4 pixels (2×2 block) into scratch.
    uint base_x = tile_x * 32u + gl_LocalInvocationID.x * 2u;
    uint base_y = tile_y * 32u + gl_LocalInvocationID.y * 2u;
    for (uint dy = 0u; dy < 2u; dy++) {
        for (uint dx = 0u; dx < 2u; dx++) {
            ivec2 uv = ivec2(int(base_x + dx), int(base_y + dy));
            vec4 px = texelFetch(current_frame_sampler, uv, 0);
            // BGRA from DMA-BUF: channel 0=B, 1=G, 2=R, 3=A (skip A).
            float ch = channel == 0u ? px.b :
                       channel == 1u ? px.g :
                                       px.r;
            uint local_x = gl_LocalInvocationID.x * 2u + dx;
            uint local_y = gl_LocalInvocationID.y * 2u + dy;
            scratch[local_y * 32u + local_x] = int(ch * 255.0);
        }
    }
    barrier();
}

void row_forward(uint row) {
    // Predict step: odd indices -= floor((left + right) / 2)
    // Each invocation handles 1 pair (its local_x within the row).
    if (gl_LocalInvocationID.y == row) {
        uint x = gl_LocalInvocationID.x * 2u + 1u; // odd index
        if (x < 32u) {
            int left  = scratch[row * 32u + (x - 1u)];
            int right = (x + 1u < 32u) ? scratch[row * 32u + (x + 1u)]
                                        : scratch[row * 32u + (x - 1u)]; // mirror
            scratch[row * 32u + x] -= (left + right) >> 1;
        }
    }
    barrier();
    // Update step: even indices += floor((left + right + 2) / 4)
    if (gl_LocalInvocationID.y == row) {
        uint x = gl_LocalInvocationID.x * 2u; // even index
        if (x < 32u) {
            int left  = (x == 0u) ? scratch[row * 32u + 1u]
                                  : scratch[row * 32u + (x - 1u)];
            int right = (x + 1u < 32u) ? scratch[row * 32u + (x + 1u)]
                                       : scratch[row * 32u + (x - 1u)]; // mirror
            scratch[row * 32u + x] += (left + right + 2) >> 2;
        }
    }
    barrier();
}

void col_forward(uint col) {
    if (gl_LocalInvocationID.x == col) {
        uint y = gl_LocalInvocationID.y * 2u + 1u;
        if (y < 32u) {
            int left  = scratch[(y - 1u) * 32u + col];
            int right = (y + 1u < 32u) ? scratch[(y + 1u) * 32u + col]
                                        : scratch[(y - 1u) * 32u + col];
            scratch[y * 32u + col] -= (left + right) >> 1;
        }
    }
    barrier();
    if (gl_LocalInvocationID.x == col) {
        uint y = gl_LocalInvocationID.y * 2u;
        if (y < 32u) {
            int left  = (y == 0u) ? scratch[32u + col]
                                  : scratch[(y - 1u) * 32u + col];
            int right = (y + 1u < 32u) ? scratch[(y + 1u) * 32u + col]
                                       : scratch[(y - 1u) * 32u + col];
            scratch[y * 32u + col] += (left + right + 2) >> 2;
        }
    }
    barrier();
}

void deinterleave_to_output(uint tile_idx, uint channel) {
    // After lifting, even indices hold low-pass (L) and odd hold high-pass (H).
    // We need to re-arrange to LL1 in top-left 16×16 quadrant, details in others.
    // Per the channel layout in encoder/cdf53.rs:
    //   work[0..16][0..16] = LL1 (we'll overwrite with deinterleave below)
    //   work[0..16][16..32] = HL1
    //   work[16..32][0..16] = LH1
    //   work[16..32][16..32] = HH1
    // For L1: each invocation writes 1 of its 4 pixels.
    uint lx = gl_LocalInvocationID.x;
    uint ly = gl_LocalInvocationID.y;
    // L_x at scratch[2*lx, 2*ly]; H_x at scratch[2*lx+1, 2*ly]
    // L_y at scratch[2*lx, 2*ly]; H_y at scratch[2*lx, 2*ly+1]
    // After both passes, the values are:
    //   scratch[2y]  [2x]   = LL (low-pass both axes)
    //   scratch[2y]  [2x+1] = HL (high-pass horizontal, low vertical)
    //   scratch[2y+1][2x]   = LH (low-horizontal, high-vertical)
    //   scratch[2y+1][2x+1] = HH

    uint channel_offset = tile_idx * COEFFS_PER_CHANNEL * CHANNELS
                        + channel * COEFFS_PER_CHANNEL;

    // Write LL1 to top-left 16×16:
    coefficients[channel_offset + ly * 16u + lx] = scratch[(ly*2u) * 32u + (lx*2u)];
    // HL1 to top-right 16×16:
    coefficients[channel_offset + 256u + ly * 16u + lx] =
        scratch[(ly*2u) * 32u + (lx*2u + 1u)];
    // LH1 to bottom-left 16×16:
    coefficients[channel_offset + 512u + ly * 16u + lx] =
        scratch[(ly*2u + 1u) * 32u + (lx*2u)];
    // HH1 to bottom-right 16×16:
    coefficients[channel_offset + 768u + ly * 16u + lx] =
        scratch[(ly*2u + 1u) * 32u + (lx*2u + 1u)];

    // NOTE: the LL1 region overlaps the LL coefficients above. They are
    // copies; L2 dispatch reads coefficients[..256] as its input.
}

void main() {
    uint compact_idx = gl_WorkGroupID.x;
    uint flat_tile_idx = tile_indices[compact_idx];
    uint tile_x = flat_tile_idx % cols;
    uint tile_y = flat_tile_idx / cols;

    for (uint channel = 0u; channel < CHANNELS; channel++) {
        load_channel_into_scratch(tile_x, tile_y, channel);
        // Horizontal pass on all 32 rows: each row handled by one invocation row.
        for (uint row = 0u; row < 32u; row++) {
            row_forward(row);
        }
        // Vertical pass on all 32 columns:
        for (uint col = 0u; col < 32u; col++) {
            col_forward(col);
        }
        deinterleave_to_output(compact_idx, channel);
    }
}
```

This is a substantial shader. It's intentionally written without inter-workgroup sync — all work for a tile happens in one workgroup with shared memory.

**NOTE for implementer:** the iteration patterns (`for row = 0..32`) are written serially per workgroup for clarity. Production-tuned versions would parallelize across the workgroup, but this is the simplest correct shape. The unit test (Step 6) verifies GPU=CPU equivalence, catching any algorithmic bugs.

```glsl
// ghostframe-lib/src/capture/shaders/cdf53_forward_l2.comp
// Stage 4c-L2: CDF 5/3 forward, level 2. Operates on the LL1 subband
// (16×16, top-left of L1's output). Output: LL2 + details in standard
// channel layout positions for L2.

#version 450
layout(local_size_x = 16, local_size_y = 16, local_size_z = 1) in;

shared int scratch[16 * 16];

layout(binding = 0, std430) buffer Cdf53Coefficients {
    int coefficients[];
};

layout(binding = 1, std430) readonly buffer Cdf53CompactList {
    uint tile_indices[];
};

layout(push_constant) uniform PushConstants {
    uint cols;
    uint rows;
    uint tile_size;
};

const uint COEFFS_PER_CHANNEL = 1024u;
const uint CHANNELS = 3u;

void main() {
    uint compact_idx = gl_WorkGroupID.x;
    // tile_idx not needed for output indexing — the coefficient buffer is
    // already keyed by compact_idx (set by L1).
    for (uint channel = 0u; channel < CHANNELS; channel++) {
        uint channel_offset = compact_idx * COEFFS_PER_CHANNEL * CHANNELS
                            + channel * COEFFS_PER_CHANNEL;
        // Load LL1 region (first 256 coefficients = first 16×16) into scratch.
        uint lx = gl_LocalInvocationID.x;
        uint ly = gl_LocalInvocationID.y;
        scratch[ly * 16u + lx] = coefficients[channel_offset + ly * 16u + lx];
        barrier();

        // 1-level CDF 5/3 lifting on the 16×16, in-place in scratch.
        // (Predict, update, deinterleave — same shape as L1 but at size=16.)
        // ... (lifting code at size 16) ...

        // Write outputs: LL2 → first 64 (8×8), HL2/LH2/HH2 → next 192.
        // Per channel layout: after L1 the LL1 was at coefficients[0..256].
        // After L2:
        //   LL2 → coefficients[0..64]   (8×8 top-left of the original LL1)
        //   HL2 → coefficients[64..128]
        //   LH2 → coefficients[128..192]
        //   HH2 → coefficients[192..256]
        // (HL1/LH1/HH1 at offsets 256/512/768 are untouched by L2.)
        // ... (write deinterleaved 8×8 sub-bands) ...
    }
}
```

**Structural notes for L2 implementation (mirrors L1 at half dimensions):**

The body of L2 is mechanically derived from L1's structure with these substitutions:
- `local_size_x = local_size_y = 16` (same as L1; we process the LL1 16×16 region inside the same workgroup).
- Loop sizes: `for (uint row = 0u; row < 16u; row++)` instead of 32. Same for cols.
- `row_forward` / `col_forward` operate on length 16 instead of 32.
- Per-invocation block size: 1×1 (one invocation per coefficient, since 16×16 = 256 = workgroup_size).
- `load_channel_into_scratch` is replaced with reading from `coefficients[channel_offset + ly*16 + lx]` (the first 256 = 16×16 LL1 from L1).
- `deinterleave_to_output` writes into 4 sub-quadrants of 64 coefficients each: LL2 → offset 0 (8×8), HL2 → 64, LH2 → 128, HH2 → 192. Per the channel layout in encoder/cdf53.rs lines documenting layout.
- The HL1/LH1/HH1 regions at coefficients[256..1024] are untouched by L2.

For brevity the full L2 body is not reproduced here. The implementer **must** copy the L1 shader file as a starting point, then apply the substitutions above. The GPU=CPU equivalence test (Step 6) verifies correctness — any algorithmic mistake produces a diff that pinpoints the first mismatching coefficient.

```rust
fn cdf53_forward_gpu_matches_cpu_reference() {
    use ghostframe_lib::encoder::cdf53;

    let mut processor = match GpuFrameProcessor::new(256) {
        Ok(p) => p,
        Err(e) => {
            eprintln!("Skipping cdf53_forward GPU=CPU test (no Vulkan?): {e}");
            return;
        }
    };

    // Construct a frame with one high-color tile (gradient) in tile (0,0)
    // and Solid elsewhere — so cdf53_forward processes exactly 1 tile.
    let width = 64u32;
    let height = 64u32;
    let stride = width * 4;
    let mut pixels = vec![0u8; (stride * height) as usize];
    for y in 0..height {
        for x in 0..width {
            let off = ((y * stride) + x * 4) as usize;
            let in_gradient = (x < 32) && (y < 32);
            if in_gradient {
                pixels[off]     = (x * 3) as u8;
                pixels[off + 1] = (y * 3) as u8;
                pixels[off + 2] = (x + y) as u8;
            } else {
                pixels[off]     = 0;
                pixels[off + 1] = 0;
                pixels[off + 2] = 0xFF;
            }
            pixels[off + 3] = 0xFF;
        }
    }

    // Extract the gradient tile's BGRA bytes for the CPU reference.
    let mut tile_bgra = vec![0u8; 32 * 32 * 4];
    for y in 0..32 {
        for x in 0..32 {
            let src = (y * stride as usize) + x * 4;
            let dst = (y * 32 + x) * 4;
            tile_bgra[dst..dst + 4].copy_from_slice(&pixels[src..src + 4]);
        }
    }
    let cpu_coefficients = cdf53::forward(&tile_bgra);

    // Drive the GPU pipeline.
    unsafe {
        let fd = make_memfd_from_pixels(&pixels, width, height);
        let result = processor.process_frame(fd, width, height, stride);
        libc::close(fd);
        let _ = match result {
            Ok(a) => a,
            Err(e) => {
                eprintln!("Skipping (memfd not DMA-BUF): {e}");
                return;
            }
        };
    }

    // Read GPU coefficients for the first (and only) Cdf53 tile from the
    // coefficient buffer. tile slot 0 = the gradient tile per cdf53_compact's
    // output order.
    let gpu_coefficients_raw = unsafe {
        std::slice::from_raw_parts(processor.cdf53_coefficients_ptr,
            cdf53::CDF53_TOTAL_COEFFS)
    };
    let gpu_coefficients: Vec<i16> = gpu_coefficients_raw.to_vec();

    // Compare exact: GPU integer lifting must match CPU integer lifting.
    for (i, (&gpu, &cpu)) in gpu_coefficients.iter().zip(cpu_coefficients.iter()).enumerate() {
        assert_eq!(gpu, cpu,
            "coefficient mismatch at index {i}: GPU={gpu} vs CPU={cpu}");
    }
}
```

```rust
// Cdf53 Phase B: read coefficients from GPU buffer, encode passes,
// enqueue 14 passes per Cdf53-classified tile.
if self.cdf53_enabled && self.client_caps.supports_cdf53 {
    let cdf53_count = unsafe { *gpu.cdf53_compact_count_ptr };
    let cdf53_list = unsafe {
        std::slice::from_raw_parts(gpu.cdf53_compact_list_ptr, cdf53_count as usize)
    };
    let coeffs_ptr = gpu.cdf53_coefficients_ptr;
    for (slot, &flat_tile_idx) in cdf53_list.iter().enumerate() {
        let tile_x = (flat_tile_idx % cols) as u8;
        let tile_y = (flat_tile_idx / cols) as u8;
        // Each slot has CDF53_TOTAL_COEFFS = 3072 i16s.
        let coefficients = unsafe {
            std::slice::from_raw_parts(
                coeffs_ptr.add(slot * crate::encoder::cdf53::CDF53_TOTAL_COEFFS),
                crate::encoder::cdf53::CDF53_TOTAL_COEFFS,
            )
        };
        let passes = crate::encoder::cdf53::encode_passes(coefficients);
        let gen = self.scheduler.bump_generation(tile_x, tile_y);
        // Diagnostic tracing: one line per pass enqueued.
        for (pass_idx, payload) in passes.iter().enumerate() {

                target: "ghostframe::cdf53",
                tile_x, tile_y, gen, pass_idx,
                payload_size = payload.len(),
                "cdf53.emit"
            );
        }
        self.scheduler.enqueue_refinement(tile_x, tile_y, gen, passes);
```
